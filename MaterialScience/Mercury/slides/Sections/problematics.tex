\begin{frame}{El Problema del Mercurio en el Ecosistema}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}[<+->]
                \item \textbf{Legado de la actividad humana:} Cientos de miles de toneladas liberadas al ambiente.
                \item \textbf{Principales fuentes:}
                \begin{itemize}
                    \item Quema de combustibles sólidos (carbón).
                    \item Minería artesanal de oro, la \alert{mayor fuente global de emisiones}.
                \end{itemize}
                \item \textbf{Bioacumulación:} Se transforma en \alert{metilmercurio} y entra en la cadena alimentaria.
            \end{itemize}
        \end{column}

        \end{column}
    \end{columns}
\end{frame}

% Diapositiva 3: Impacto en la salud
\begin{frame}{Impacto en la Salud Humana}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}[<+->]
                \item \textbf{Vía de exposición:} Principalmente a través del consumo de \alert{pescado y marisco}.
                \item \textbf{Biomagnificación:} Los grandes depredadores marinos (atún, pez espada) acumulan las concentraciones más altas.
                \item \textbf{Efectos en la salud:}
                \begin{itemize}
                    \item Daños al sistema nervioso y órganos vitales.
                    \item \alert{Riesgo crítico} para el feto y niños pequeños, afectando el desarrollo cerebral.
                \end{itemize}
            \end{itemize}
        \end{column}

    \end{columns}
\end{frame}

% Diapositiva 4: Acciones globales
\begin{frame}{Acciones Globales para Abordar el Problema}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}[<+->]
                \item \textbf{El Convenio de Minamata:} Primer acuerdo internacional para abordar la contaminación por mercurio.
                \item \textbf{Objetivos:} Controlar las emisiones, reducir el uso y gestionar su eliminación de forma segura.
                \item \textbf{Desafío:} Las emisiones de una región pueden viajar largas distancias, por lo que la \alert{cooperación internacional} es crucial.
            \end{itemize}
        \end{column}
