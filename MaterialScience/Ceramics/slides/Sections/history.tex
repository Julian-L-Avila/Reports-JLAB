\section{Historia Pre Moderna}

\begin{frame}{Primeros descubrimientos ($\sim$28,000 BCE - 1750 CE)}
    \begin{itemize}[<+->]
        \item \textbf{Los comienzos:} Los primeros hallazgos (Anatolia) son figuras de arcilla no cocida.
        \item \textbf{El gran salto:} El descubrimiento accidental de que el fuego transforma la arcilla en un material inalterable.
        \item \textbf{La revolución:} La invención del \alert{torno de alfarero} en Mesopotamia (3000-4000 BCE) acelera y estandariza la producción.
    \end{itemize}

        \caption{Torno de alfarero primitivo. \cite{Arazola}}
    \end{figure}
\end{frame}

% Diapositiva 3: Avances tecnológicos
\begin{frame}{Avances tecnológicos clave ($\sim$4500 BCE - 1750 CE)}
    \begin{itemize}[<+->]
        \item \textbf{El vidriado:} Desarrollado en Mesopotamia, se inventó al intentar replicar la piedra lapislázuli.
        \item \textbf{¿Por qué es importante?} Hace que la cerámica sea impermeable y le da un acabado brillante y atractivo.
    \end{itemize}

        \centering

        \caption{Jarrón de la Alhambra. \cite{WikiwandAlhambra}}
    \end{figure}
\end{frame}

% Diapositiva 4: La cerámica como arte y producción
\begin{frame}{La cerámica como lienzo ($\sim$1000 BCE - 1750 CE)}
    \begin{itemize}[<+->]
        \item \textbf{Grecia:} La cerámica se convierte en un medio para contar historias. Se perfeccionan técnicas como las \alert{figuras negras} y \alert{figuras rojas}.
        \item \textbf{Roma:} Se enfocan en la \alert{producción en masa}. Desarrollan la \alert{cerámica de brillo rojo} para una distribución amplia en el Imperio.
    \end{itemize}
    \begin{figure}
        \centering
        \includegraphics[width=0.4\linewidth]{Figures/Rojo-Grecia.jpg}
        \caption{Cerámica griega de figuras rojas. \cite{Griegos}}
    \end{figure}
\end{frame}

% Diapositiva 5: Influencia asiática
\begin{frame}{Influencia asiática y el auge de la porcelana (hasta 1750 CE)}
    \begin{itemize}[<+->]
        \item \textbf{La porcelana:} La invención de la porcelana en China introdujo un nuevo estándar de finura y translucidez que fascinó a Europa.
        \item \textbf{Ejemplo:} Fue imitada en las cerámicas de \alert{Delft} en Holanda.
    \end{itemize}
